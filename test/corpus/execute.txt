Execute with into and using

create function f(tbl text) returns int language plpgsql as $$
declare
  n int;
  r record;
begin
  execute 'select count(*) from ' || quote_ident(tbl) into n;
  execute format('insert into %I values ($1)', tbl) using n;
  execute 'select 1, 2' into strict n, r.b;
  execute 'select $1 + $2' into n using 1, 2;
  return n;
end;
$$;

declare
  r record;
begin
  for r in select id from t loop
    return next r.id;
  end loop;
  return query select id from t;
  return query execute 'select 1' using 2;
  return;
end;

Select into strict and returning into
================================================================================

create function f() returns int language plpgsql as $$
declare
  n int;
  r record;
begin
  select count(*) into n from t;
  select count(*) into strict n from t;
  select a, b into r.a, r.b from t;
  insert into t values (1) returning id into n;
  update t set a = 1 where id = 1 returning a, b into strict r.a, r.b;
  delete from t where id = 1 returning * into r;
  return n;
end;
$$;

Open a cursor for a dynamic query
================================================================================

create function f() returns void language plpgsql as $$
declare
  refc refcursor;
begin
  open refc for execute 'select 1';
  open refc scroll for execute format('select * from %I', 't') using 1;
  close refc;
end;
$$;
