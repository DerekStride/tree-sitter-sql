================================================================================
Open, fetch and close a cursor
================================================================================

create function f() returns int language plpgsql as $$
declare
  cur cursor for select id from t;
  cur2 cursor (key int) for select id from t where id = key;
  v int;
begin
  open cur;
  fetch cur into v;
  fetch next from cur into v;
  fetch prior in cur into v;
  fetch absolute 3 from cur into v;
  fetch relative -1 from cur into v;
  fetch first from cur into r.a, r.b;
  move last in cur;
  move cur;
  close cur;
  open cur2(5);
  return v;
end;
$$;

  refc refcursor;
begin
  open refc for select 1;
  open refc scroll for select * from t where a = 1;
  open refc no scroll for (select 1);
  close refc;

================================================================================
Cursor loop in an anonymous block
================================================================================

DECLARE
  CURSOR c_emp IS SELECT employee_id, last_name FROM employees WHERE department_id = 10;
  l_rec c_emp%ROWTYPE;
BEGIN
  OPEN c_emp;
  LOOP
    FETCH c_emp INTO l_rec;
    EXIT WHEN l_rec.employee_id IS NULL;
  END LOOP;
  CLOSE c_emp;
END;
