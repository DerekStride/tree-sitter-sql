================================================================================
Meta-command between statements
================================================================================

CREATE DATABASE analytics;

\c analytics

CREATE TABLE events (id BIGINT NOT NULL PRIMARY KEY);

--------------------------------------------------------------------------------

(program
  (statement
    (create_database
      (keyword_create)
      (keyword_database)
      (identifier)))
  (meta_command)
  (statement
    (create_table
      (keyword_create)
      (keyword_table)
      (object_reference
        name: (identifier))
      (column_definitions
        (column_definition
          name: (identifier)
          type: (bigint
            (keyword_bigint))
          (keyword_not)
          (keyword_null)
          (keyword_primary)
          (keyword_key))))))
